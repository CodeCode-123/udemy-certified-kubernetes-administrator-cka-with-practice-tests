# Manual Scheduling
1. Add the nodeName: node-name to the Pod spec, e.g.
---
apiVersion: v1
kind: Pod
metadata:
  name: nginx
  labels:
    name: nginx
spec:
  containers:
  - name: nginx
    image: nginx
    ports:
    - containerPort: 8080
  nodeName: node02
---



# Labels and Selectors
1. Add labels with the key: value pairs in the metadata of the Pod manifest file, e.g.
---
apiVersion: v1
kind: Pod
metadata:
  name: simple-webapp
  labels:
    app: app1
    function: front-end
spec:
  containers:
    - name: simple-webapp
      image: simple-webapp
      ports:
        - containerPort: 8080
---

2. Use kubectl to add labels
kubectl label pods <pod-name> env=prod tier=backend version=1.2

3. Select labeled objects
kubectl get pods --selector app=app1
kubectl get pods -l app=app1,function=front-end



# Taints and Tolerations
1. Add a taint to a node
kubectl taint nodes <node-name> <key>=<value>:<effect>
kubectl taint node node1 app=blue:NoSchedule

2. Delete a taint from a node
kubectl taint node node1 app=blue:NoSchedule-

3. Check taints
kubectl describe node node1 | grep Taints



# Node Selectors
1. Add nodeSelector: key: value to the Pod manifest file, e.g.
---
apiVersion: v1
kind: Pod
metadata:
  name: myapp-pod
spec:
  containers:
    - name: data-processor
      image: data-processor
  nodeSelector:
    size: large
---

2. Add a label to the node
kubectl label nodes node01 size=large

3. Verify the label
kubectl get node node01 --show-labels



# Node Affinity
1. Add affinity to the Pod manifest file, e.g.
---
apiVersion: v1
kind: Pod
metadata:
  name: myapp-pod
spec:
  containers:
    - name: data-processor
      image: data-processor
  affinity:
    nodeAffinity:
      requiredDuringSchedulingIgnoredDuringExecution:
        nodeSelectorTerms:
          - matchExpressions:
              - key: size
                operator: In
                values:
                  - Large
---

If only a key is used without a value, the operator is set to Exists, e.g.

matchExpressions:
  - key: size
    operator: Exists



# Taints and Tolerations vs Node Affinity, Resource Requirements
1. Combination of using Taints, Tolerations, and Node Affinity can assign pods to particular nodes

2. Resource Requests (minimum), Limits (maximum), add resources: requests: ... and limits: ... to the Pod spec.containers, e.g.
---
apiVersion: v1
kind: Pod
metadata:
  name: simple-webapp-color
  labels:
    name: simple-webapp-color
spec:
  containers:
    - name: simple-webapp-color
      image: simple-webapp-color
      ports:
        - containerPort: 8080
      resources:
        requests:
          memory: "1Gi"
          cpu: "2"
        limits:
          memory: "2Gi"
          cpu: "2"
---

3. LimitRanges can be used for all pods in a particular namespace, e.g.
---
apiVersion: v1
kind: LimitRange
metadata:
  name: cpu-resource-constraint
spec:
  limits:
    - type: Container
      default:
        cpu: 500m
      defaultRequest:
        cpu: 500m
      max:
        cpu: "1"
      min:
        cpu: 100m
---
apiVersion: v1
kind: LimitRange
metadata:
  name: memory-resource-constraint
spec:
  limits:
    - type: Container
      default:
        memory: 1Gi
      defaultRequest:
        memory: 1Gi
      max:
        memory: 1Gi
      min:
        memory: 500Mi
---

4. ResourceQuota can be used to limit the total resources in a namespace, e.g.
---
apiVersion: v1
kind: ResourceQuota
metadata:
  name: my-resource-quota
spec:
  hard:
    requests.cpu: "4"
    requests.memory: "4Gi"
    limits.cpu: "10"
    limits.memory: "10Gi"
---

5. Use kubectl describe pod <pod-name> to check the resources of a pod
kubectl describe pod rabbit

6. Check the YAML configuration
kubectl get pod <pod-name> -n <namespace> -o yaml
kubectl get pod <pod-name> -n <namespace> -o yaml > <pod-manifest>.yaml

kubectl get pod rabbit -n default -o yaml
kubectl get pod rabbit -n default -o yaml > rabbit-pod.yaml

# Static Pods
1. Find the static pods
kubectl get pods --all-namespaces //the name ends with -node-name
kubectl describe pods -A | grep -E "^Name:|^Namespace:|Controlled By:" //the static pods are controlled by "Node/node-name"

2. Find the path holding static pod manifests
ps aux | grep kubelet //find the --config flag and path, usually /var/lib/kubelet/config.yaml, or find the pod-manifest-path
grep -i staticPodPath /var/lib/kubelet/config.yaml

3. Create a static pod
ssh node-name //ssh to the node to create the pod
kubectl run static-busybox --image=busybox --dry-run=client -o yaml --command -- sleep 1000 > /etc/kubernetes/manifests/static-busybox.yaml
kubectl run static-busybox --image=busybox --command sleep 1000 --dry-run=client -o yaml > /etc/kubernetes/manifests/stati
c-busybox.yaml //use --dry-run=client to create a static pod, if no --dry-run=client,
// the pod will not be static even if created in /etc/kubernetes/manifests
(create a pod: kubectl run pod-name; kubectl create -f pod-name.yaml;
create other resources: kubectl create deployment; service; configmap; secret; namespace; quota, and so on)

4. Delete a static pod
ssh node-name //ssh to the node to delete the pod
ps aux | grep kubelet //find the --config path
grep -i staticPodPath /path/config.yaml //find the staticPodPath containing the pod manifest file
ls /staticPodPath //find the pod manifest file
rm -f /staticPodPath/pod-manifest.yaml
exit //exit this node and return to the default node



# Priority Classes
//Kubernetes automatically created two built-in PriorityClass objects: system-node-critical and system-cluster-critical
kubectl get priorityclasses //get all priorityclasses
//If you set the preemptionPolicy: PreemptLowerPriority, the pod with a higher value will evict the pod with a lower value
//create a priority class
kubectl create priorityclass high-priority --value=100000 --preemption-policy="PreemptLowerPriority" -o yaml > high-priority.yaml
//create a pod with a priority class
kubectl run low-prio-pod --image=nginx --dry-run=client -o yaml > low-prio-pod.yaml
//then set the spec.priorityClassName: <priority-name>
kubectl create -f low-prio-pod.yaml

kubectl get pods -o custom-columns="NAME:.metadata.name,PRIORITY:.spec.priorityClassName" //check the priority class of all pods
kubectl describe pod <pod-name> //get the details of a pod, including the priorityClassName

//Create a scheduler using a Pod manifest file and spec.containers[0].command
apiVersion: v1
kind: Pod
metadata:
  name: my-scheduler
  namespace: kube-system
spec:
  containers:
  - name: kube-scheduler
    image: k8s.gcr.io/kube-scheduler-amd64:v1.22.0
    command:
      - kube-scheduler
      - --address=127.0.0.1
      - --kubeconfig=/etc/kubernetes/scheduler.conf
      - --config=/etc/kubernetes/my-scheduler-config.yaml

//The scheduler config yaml with leaderElection
apiVersion: kubescheduler.config.k8s.io/v1
kind: KubeSchedulerConfiguration
profiles:
- schedulerName: my-scheduler
leaderElection:
  leaderElect: true
  resourceNamespace: kube-system
  resourceName: lock-object-my-scheduler

# Configuring Scheduler Profiles
kubectl describe node <node-name> //check the Unschedulable is true or false
//Scheduler plugins:
  Scheduling Queue: PrioritySort
  Filtering: NodeResourcesFit, NodeName, NodeUnschedulable, TaintToleration, NodePorts, NodeAffinity
  Scoring: NodeResourcesFit, ImageLocality, TaintToleration, NodeAffinity
  Binding: DefaultBinder
//Each pod is scheduled by one scheduler; if not defined in the pod yaml file, the pod is scheduled by the default scheduler
//Set a customized scheduler to a pod, e.g.
---
apiVersion: v1
kind: Pod
metadata:
  name: custom-scheduled-pod
spec:
  schedulerName: my-scheduler
  containers:
  - name: app
    image: nginx
---

//Turn off and turn on some plugins of a scheduler: e.g. score plugin
---
apiVersion: kubescheduler.config.k8s.io/v1
kind: KubeSchedulerConfiguration
profiles:
  - schedulerName: default-scheduler
    plugins:
      score:
        disabled:
          - name: ImageLocality       # Disables scoring based on local container images
          - name: PodTopologySpread   # Disables default topology spread scoring
          # - name: "*"               # Use "*" to disable ALL default score plugins
        enabled:
          - name: NodeResourcesFit
            weight: 2
---
